rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId} {
      allow read, create, update: if request.auth != null && request.auth.uid == userId;
      allow delete: if false;

      match /notifications/{notificationId} {
        allow read: if request.auth != null && request.auth.uid == userId;
        allow update: if request.auth != null
          && request.auth.uid == userId
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['read']);
        allow create, delete: if false;
      }
    }

    match /campaigns/{campaignId} {
      allow read: if request.auth != null && (
        resource.data.ownerFirebaseUid == request.auth.uid ||
        resource.data.owner_firebase_uid == request.auth.uid ||
        resource.data.ownerId == request.auth.uid ||
        resource.data.workspaceId == request.auth.uid ||
        resource.data.ownerEmail == request.auth.token.email ||
        resource.data.authorEmail == request.auth.token.email ||
        resource.data.userId == request.auth.uid ||
        resource.data.createdBy == request.auth.uid ||
        resource.data.status == 'active' ||
        !('ownerFirebaseUid' in resource.data)
      );
      allow create: if request.auth != null && (
        request.resource.data.ownerFirebaseUid == request.auth.uid ||
        request.resource.data.owner_firebase_uid == request.auth.uid ||
        request.resource.data.workspaceId == request.auth.uid ||
        request.resource.data.ownerEmail == request.auth.token.email
      );
      allow update: if request.auth != null && (
        resource.data.ownerFirebaseUid == request.auth.uid ||
        resource.data.owner_firebase_uid == request.auth.uid ||
        resource.data.ownerId == request.auth.uid ||
        resource.data.workspaceId == request.auth.uid ||
        resource.data.ownerEmail == request.auth.token.email ||
        resource.data.authorEmail == request.auth.token.email ||
        resource.data.userId == request.auth.uid ||
        resource.data.createdBy == request.auth.uid ||
        (
          !('ownerFirebaseUid' in resource.data) &&
          request.resource.data.ownerFirebaseUid == request.auth.uid &&
          request.resource.data.ownerEmail == request.auth.token.email
        ) ||
        (
          !('ownerEmail' in resource.data) &&
          request.auth.token.email != null &&
          request.resource.data.ownerFirebaseUid == request.auth.uid &&
          request.resource.data.ownerEmail == request.auth.token.email
        )
      );
      allow delete: if request.auth != null && (
        resource.data.ownerFirebaseUid == request.auth.uid ||
        resource.data.owner_firebase_uid == request.auth.uid ||
        resource.data.ownerId == request.auth.uid ||
        resource.data.workspaceId == request.auth.uid ||
        resource.data.ownerEmail == request.auth.token.email
      );
    }

    match /submissions/{submissionId} {
      allow read: if request.auth != null && (
        resource.data.creatorFirebaseUid == request.auth.uid ||
        resource.data.campaignOwnerUid == request.auth.uid
      );
      allow create: if request.auth != null &&
        request.resource.data.creatorFirebaseUid == request.auth.uid &&
        request.resource.data.status == 'pending';
      allow update: if request.auth != null &&
        resource.data.campaignOwnerUid == request.auth.uid;
      allow delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
